### 2) 新增 `lbOutpostInfo(cls)`（仿 `lbNoteInfo` 模式，一处定义）

```js
var LB_OUTPOST_CAT = [['Attack_Training','训练'],['Attack_Tame','驯服'],['Attack_Ranged','远程'],
                      ['Attack_Industrial','工业'],['Defend','防守'],['ResourceCache','资源缓存']];
var LB_OUTPOST_SZ   = [['Small','小'],['Medium','中'],['Large','大'],['City','城区']];
var LB_OUTPOST_TIER = [['Alpha','α','#3b82f6'],['Beta','β','#facc15'],['Gamma','γ','#ef4444']];   // 蓝/黄/红（与洞穴箱 1/2/3 同口径）
var LB_OUTPOST_ORD  = ['任务·α','任务·β','任务·γ'];
function lbOutpostInfo(cls){
  var c = String(cls || '');
  var t = null; for(var i=0;i<LB_OUTPOST_TIER.length;i++){ if(c.indexOf('_'+LB_OUTPOST_TIER[i][0])>=0){ t = LB_OUTPOST_TIER[i]; break; } }
  var cat=''; for(var j=0;j<LB_OUTPOST_CAT.length;j++){ if(c.indexOf(LB_OUTPOST_CAT[j][0])>=0){ cat = LB_OUTPOST_CAT[j][1]; break; } }
  var sz='';  for(var k=0;k<LB_OUTPOST_SZ.length;k++){ if(c.indexOf(LB_OUTPOST_SZ[k][0])>=0){ sz = LB_OUTPOST_SZ[k][1]; break; } }
  return { g:[t ? ('任务·'+t[1]) : '任务', (t ? t[2] : '#94a3b8')], label:(cat||'任务') + (sz ? '·'+sz : '') + (t ? '·'+t[1] : '') };
}
```
